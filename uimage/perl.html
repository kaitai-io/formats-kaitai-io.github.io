U-Boot Image wrapper: Perl parsing library

The legacy U-Boot image format (uImage), as created by mkimage: a 64-byte header followed by the image data (e.g. a Linux kernel, a ramdisk or a script). The header describes the data (OS, CPU architecture, image type, compression, load and entry point addresses, name) and protects both the header and the data with CRC-32 checksums.

U-Boot's newer FIT (Flattened Image Tree) format is not covered by this spec.

    my ($self) = @_;

    $self->{data} = $self->{_io}->read_bytes($self->header()->len_image());
}

sub new {

sub _read {
    my ($self) = @_;

    if (!($self->has_asus_info())) {
        $self->{name} = Encode::decode("UTF-8", $self->{_io}->read_bytes_term(0, 0, 1, 0));
    }
    if ($self->has_asus_info()) {
        $self->{asus_info} = Uimage::AsusFirmwareInformation->new($self->{_io}, $self, $self->{_root});
    }
}

sub byte0 {
    my ($self) = @_;
    return $self->{byte0} if ($self->{byte0});
    my $_pos = $self->{_io}->pos();
    $self->{_io}->seek(0);
    $self->{byte0} = $self->{_io}->read_u1();
    $self->{_io}->seek($_pos);
    return $self->{byte0};
}

sub byte4 {
    my ($self) = @_;
    return $self->{byte4} if ($self->{byte4});
    my $_pos = $self->{_io}->pos();
    $self->{_io}->seek(4);
    $self->{byte4} = $self->{_io}->read_u1();
    $self->{_io}->seek($_pos);
    return $self->{byte4};
}

sub has_asus_info {
    my ($self) = @_;
    return $self->{has_asus_info} if ($self->{has_asus_info});
    $self->{has_asus_info} =  (($self->byte0() < 32) && ($self->byte4() >= 32) && ($self->byte4() < 127)) ;
    return $self->{has_asus_info};
}

sub name {
    my ($self) = @_;
    return $self->{name};
}

sub asus_info {
    my ($self) = @_;
    return $self->{asus_info};
}
