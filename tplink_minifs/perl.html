TP-Link MiniFS: Perl parsing library

MiniFS is a file system found in certain firmware files from TP-Link (including its Mercusys brand) and Mercury. Such firmware contains an RTOS kernel, e.g. VxWorks in RE450(V4)_210531.zip (archived) or TPOS, TP-Link's RTOS, in TL-WA850RE(EU)_V7.0_250808.zip (archived).

According to a TP-Link manager quoted in a 2021 article (archived), Mercusys has been sold in China as Mercury since 2001. All Mercury images examined contain a certificate with O=tplink in its issuer and subject.

All versions share the same header, and its version field determines the layout of the rest of the image.

In version 3, the header is followed by a table of directory paths and file names, followed by meta information for each file, meta information about the LZMA compressed blobs including length, offset and uncompressed length, and then a number of LZMA compressed blobs.

The LZMA blobs can contain the concatenated contents of multiple files. The meta information of a file contains the offsets of its directory path and file name, the LZMA blob (numbering starts at 0), the offset of the file in the uncompressed data and the size of the uncompressed file.

Some version 3 images have no uncompressed length in the meta information about the blobs (see has_8_byte_lzma_metas in body_v3).

In version 2, the header is followed by the length of the files area and a meta information record for each file, each followed directly by the file's data. Unlike versions 1 and 3, version 2 stores no paths or file names at all, only the MD5 hash of each file's path.

In version 1, the header is followed by a record with the path, the compressed length and the offset of each file, and then one LZMA blob per file.

This Kaitai Struct implementation was written independently of the paper Reverse Engineered MiniFS File System. It was started in the firmware analysis project BANG (Binary Analysis Next Generation) in 2021 and submitted to the Kaitai Struct format gallery in September 2023, whereas the paper was published in July 2024.

    my ($class, $_io, $_parent, $_root) = @_;
    my $self = IO::KaitaiStruct::Struct->new($_io);

    bless $self, $class;
    $self->{_parent} = $_parent;
    $self->{_root} = $_root || $self;

    $self->_read();

    return $self;
}

sub _read {
    my ($self) = @_;

    $self->{header} = TplinkMinifs::MinifsHeader->new($self->{_io}, $self, $self->{_root});
    my $_on = $self->header()->version();
    if ($_on == 1) {
        $self->{body} = TplinkMinifs::BodyV1->new($self->{_io}, $self, $self->{_root});
    }
    elsif ($_on == 2) {
        $self->{body} = TplinkMinifs::BodyV2->new($self->{_io}, $self, $self->{_root});
    }
    elsif ($_on == 3) {
        $self->{body} = TplinkMinifs::BodyV3->new($self->{_io}, $self, $self->{_root});
    }
}

sub new {
    my ($class, $_io, $_parent, $_root) = @_;
    my $self = IO::KaitaiStruct::Struct->new($_io);

    $self->{_parent} = $_parent;
    $self->{_root} = $_root;

    return $self;

sub _read {
    my ($self) = @_;

    $self->{len_name_table} = $self->{_io}->read_u4be();
    $self->{_raw_name_table} = $self->{_io}->read_bytes($self->len_name_table());
    my $io__raw_name_table = IO::KaitaiStruct::Stream->new($self->{_raw_name_table});
    $self->{name_table} = TplinkMinifs::NameTable->new($io__raw_name_table, $self, $self->{_root});
    $self->{inodes} = [];
    my $n_inodes = $self->_root()->header()->num_files();
    for (my $i = 0; $i < $n_inodes; $i++) {
        push @{$self->{inodes}}, TplinkMinifs::Inode->new($self->{_io}, $self, $self->{_root});
    }
    if ($self->ofs_lzma_metas() < 0) {
        $self->{_unnamed3} = $self->{_io}->read_bytes(0);
    }
    $self->{lzma_metas} = [];
    my $n_lzma_metas = $self->num_lzma_blobs();
    for (my $i = 0; $i < $n_lzma_metas; $i++) {
        push @{$self->{lzma_metas}}, TplinkMinifs::LzmaMeta->new($self->{_io}, $self, $self->{_root});
    }
    $self->{_raw_lzma_blobs_area} = $self->{_io}->read_bytes(@{$self->lzma_metas()}[-1]->ofs_end());
    my $io__raw_lzma_blobs_area = IO::KaitaiStruct::Stream->new($self->{_raw_lzma_blobs_area});
    $self->{lzma_blobs_area} = TplinkMinifs::Dummy->new($io__raw_lzma_blobs_area, $self, $self->{_root});
}

sub has_8_byte_lzma_metas {
    my ($self) = @_;
    return $self->{has_8_byte_lzma_metas} if ($self->{has_8_byte_lzma_metas});
    $self->{has_8_byte_lzma_metas} = $self->len_first_blob_uncompressed_if_8_byte_metas() == 18446744073709551615;
    return $self->{has_8_byte_lzma_metas};
}

sub len_first_blob_uncompressed_if_8_byte_metas {
    my ($self) = @_;
    return $self->{len_first_blob_uncompressed_if_8_byte_metas} if ($self->{len_first_blob_uncompressed_if_8_byte_metas});
    my $_pos = $self->{_io}->pos();
    $self->{_io}->seek(($self->ofs_lzma_metas() + 8 * $self->num_lzma_blobs()) + 5);
    $self->{len_first_blob_uncompressed_if_8_byte_metas} = $self->{_io}->read_u8le();
    $self->{_io}->seek($_pos);
    return $self->{len_first_blob_uncompressed_if_8_byte_metas};
}

sub len_lzma_meta {
    my ($self) = @_;
    return $self->{len_lzma_meta} if ($self->{len_lzma_meta});
    $self->{len_lzma_meta} = ($self->has_8_byte_lzma_metas() ? 8 : 12);
    return $self->{len_lzma_meta};
}

sub num_lzma_blobs {
    my ($self) = @_;
    return $self->{num_lzma_blobs} if ($self->{num_lzma_blobs});
    $self->{num_lzma_blobs} = @{$self->inodes()}[-1]->cur_max_lzma_blob() + 1;

}

package TplinkMinifs::NameTable;

our @ISA = 'IO::KaitaiStruct::Struct';

sub from_file {
    my ($class, $filename) = @_;
    my $fd;

    open($fd, '<', $filename) or return undef;
    binmode($fd);
    return new($class, IO::KaitaiStruct::Stream->new($fd));
}

sub new {
    my ($class, $_io, $_parent, $_root) = @_;
    my $self = IO::KaitaiStruct::Struct->new($_io);

    bless $self, $class;
    $self->{_parent} = $_parent;
    $self->{_root} = $_root;

    $self->_read();

    return $self;
}

sub _read {
    my ($self) = @_;

    $self->{names} = [];
    while (!$self->{_io}->is_eof()) {
        push @{$self->{names}}, Encode::decode("ASCII", $self->{_io}->read_bytes_term(0, 0, 1, 1));
    }
}

sub names {
    my ($self) = @_;
    return $self->{names};
}
